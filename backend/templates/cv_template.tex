% ── Custom commands ───────────────────────────────────────
% Experience entry: \expentry{Title}{Date}{Company}{Location}{bullets}
\newcommand{\expentry}[5]{
  \noindent\textbf{#1}\hfill\textit{#2}\\
  \textit{#3}{\small\ #4}\\[2pt]
  \begin{itemize}[nosep,leftmargin=1.5em,topsep=0pt,partopsep=0pt]
    #5
  \end{itemize}
  \vspace{4pt}
}

% Education entry: \eduentry{School}{Date}{Degree}{details}
\newcommand{\eduentry}[4]{
  \noindent\textbf{#1}\hfill\textit{#2}\\
  \textit{#3}\\[2pt]
  #4
  \vspace{4pt}
}

% Project entry: \projentry{Name}{Date}{Description}{bullets}

    #4
  \end{itemize}
  \vspace{4pt}
}
